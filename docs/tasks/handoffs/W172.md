```text
import sys; sys.path.insert(0, ".")
import re
from pathlib import Path
from tools.quality import config
from tools.quality.handoffs import (HANDOFF_DIR, RULING_RECORD, TASK_HANDOFF,
                                    OFFICE_HANDOFF, declared_kind)
from tools.quality.handoffs.contract import (_FINDING_NUMBER, _MARKUP,
                                             _claim_free_end, marker_lines)
from tools.quality.handoffs.records import record_scopes

root, universe, records = Path("."), set(), []
for path in sorted((root / HANDOFF_DIR).rglob("*.md")):
    rel, text = config.relative(path, root), config.read_text(path)
    if text is None:
        continue
    kind, ids, _ = declared_kind(text)
    stem = rel.rsplit("/", 1)[-1].removesuffix(".md")
    if kind in (TASK_HANDOFF, OFFICE_HANDOFF):
        universe.update(ids)
    elif kind == RULING_RECORD:
        scopes = record_scopes(stem, ids)
        universe.update(scopes)
        records.append((rel, text, scopes))
for path in sorted((root / "docs/tasks/rows").glob("*.md")):
    universe.add(path.stem)
for path in sorted((root / "docs/tasks").glob("E*.md")):
    universe.update(re.findall(r"`([A-Z]{2,4}-\d{2}[a-z]?)`",
                               config.read_text(path) or ""))

cited, dangling = 0, []
for rel, text, scopes in records:
    for number, line, own_line in marker_lines(text):
        if not own_line:
            continue
        claim = _FINDING_NUMBER.match(line[_claim_free_end(line, _MARKUP):])
        if claim is None or claim.group("scope") is None:
            continue
        scope = claim.group("scope")
        if scope in scopes:
            continue
        cited += 1
        if scope not in universe:
            dangling.append((rel, number, scope))
print(len(universe), "scopes;", cited, "citations;", len(dangling), "dangling")
```
